cmake_minimum_required(VERSION 3.16)
project(transime_native_worker_tests LANGUAGES CXX)
find_package(Fcitx5Utils CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c>=0.16)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(transime_worker_client_test client_test.cpp ../../plugin/worker_client.cpp)
target_compile_features(transime_worker_client_test PRIVATE cxx_std_20)
target_include_directories(transime_worker_client_test PRIVATE ../../plugin ../../core)
target_compile_options(transime_worker_client_test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(transime_worker_client_test PRIVATE Fcitx5::Utils PkgConfig::JSONC)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/empty-model")
add_test(NAME native_worker_transport COMMAND transime_worker_client_test
    "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fake_worker.py" "${CMAKE_BINARY_DIR}/empty-model")
set_tests_properties(native_worker_transport PROPERTIES TIMEOUT 20)
