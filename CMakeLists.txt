cmake_minimum_required(VERSION 3.16)
project(transime VERSION 0.1.4 LANGUAGES CXX)

# Dependency-free state core. The optional Fcitx module is built separately.
add_library(transime_core STATIC core/session.cpp)
target_include_directories(transime_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/core")
target_compile_features(transime_core PUBLIC cxx_std_17)
set_target_properties(transime_core PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(transime_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(TRANSIME_BUILD_FCITX "Build the experimental Fcitx module against an isolated SDK" OFF)
option(TRANSIME_ENABLE_TEST_FIXTURE "Compile synthetic translations for integration tests only" OFF)
option(TRANSIME_BUILD_HEADLESS_TESTS "Build the isolated input test frontend (requires static keyboard library)" OFF)
set(TRANSIME_FCITX_VERSION "5.1.19" CACHE STRING "Exact Fcitx SDK version for this build")
if(TRANSIME_BUILD_FCITX)
    find_package(Fcitx5Core ${TRANSIME_FCITX_VERSION} EXACT REQUIRED)
    find_package(Fcitx5Config ${TRANSIME_FCITX_VERSION} EXACT REQUIRED)
    add_subdirectory(plugin)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(transime_core_test tests/core_test.cpp)
    target_link_libraries(transime_core_test PRIVATE transime_core)
    set_target_properties(transime_core_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(transime_core_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME transime_core_invariants COMMAND transime_core_test)
    add_executable(transime_translation_cache_test tests/test_translation_cache.cpp)
    target_link_libraries(transime_translation_cache_test PRIVATE transime_core)
    set_target_properties(transime_translation_cache_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(transime_translation_cache_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME transime_translation_cache_invariants COMMAND transime_translation_cache_test)
    if(TRANSIME_BUILD_FCITX AND TRANSIME_BUILD_HEADLESS_TESTS)
        add_subdirectory(tests/fcitx)
    endif()
endif()
